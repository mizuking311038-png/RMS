# Improved checks avoiding inline quoting problems
$root = Get-Location
Write-Output "Root: $root"
$htmlFiles = Get-ChildItem -Recurse -File -Include *.html,*.htm
$external = @(); $relativeMissing = @(); $anchorMissing = @(); $imgMissingAlt = @(); $emptyHref = @();
foreach ($f in $htmlFiles) {
  $text = Get-Content $f.FullName -Raw -ErrorAction SilentlyContinue
  if (-not $text) { continue }
  
  # parse href/src simply
  $re = 'href\s*=\s*(?:"([^"]*)"|\'([^\']*)\')'
  $m = [regex]::Matches($text, $re)
  foreach ($mm in $m) { $link = if ($mm.Groups[1].Value) { $mm.Groups[1].Value } else { $mm.Groups[2].Value }; $clean = $link.Split('?')[0].Split('#')[0]; if ($clean -match '^(http:|https:)') { $external += @{file=$f.FullName; link=$link} } elseif ($clean -match '^#') { $anchor = $clean.Substring(1); if (-not ($text -match "(id|name)\s*=\s*['\"]$anchor['\"]")) { $anchorMissing += @{file=$f.FullName; anchor=$anchor}} } else { if ($clean -match '^/') { $target = Join-Path $root ($clean.TrimStart('/')) } else { $target = Join-Path $f.DirectoryName $clean } if (-not (Test-Path $target)) { $tryIndex = Join-Path $target 'index.html'; if (-not (Test-Path $tryIndex)) { $relativeMissing += @{file=$f.FullName; link=$link; target=$target} } } }
  }

  $re2 = 'src\s*=\s*(?:"([^"]*)"|\'([^\']*)\')'
  $m2 = [regex]::Matches($text, $re2)
  foreach ($mm in $m2) { $link = if ($mm.Groups[1].Value) { $mm.Groups[1].Value } else { $mm.Groups[2].Value }; $clean = $link.Split('?')[0].Split('#')[0]; if ($clean -match '^(http:|https:)') { $external += @{file=$f.FullName; link=$link} } else { if ($clean -match '^/') { $target = Join-Path $root ($clean.TrimStart('/')) } else { $target = Join-Path $f.DirectoryName $clean } if (-not (Test-Path $target)) { $tryIndex = Join-Path $target 'index.html'; if (-not (Test-Path $tryIndex)) { $relativeMissing += @{file=$f.FullName; link=$link; target=$target} } } }
  }

  if ($text -match '')[0]) + '>'; if ($tag -notmatch 'alt\s*=') { $imgMissingAlt += @{file=$f.FullName; tag=$tag} } } }
  if ($text -match 'href\s*=\s*"\s*"|href\s*=\s*\\'\s*\\'') { $emptyHref += @{file=$f.FullName} }
}

# Check externals
$externalResults = @(); if ($external.Count -gt 0) { $seen = @{}; foreach ($e in $external) { $url = $e.link; if ($seen.ContainsKey($url)) { continue } $seen[$url]=1; try { $code = & curl.exe -I -s -o NUL -w "%{http_code}" $url 2>$null } catch { $code = 'ERR' } $externalResults += @{link=$url; status=$code} } }

Write-Output "Scanned HTML: $($htmlFiles.Count)"; Write-Output "External: $($external.Count)"; Write-Output "Relative missing: $($relativeMissing.Count)"; Write-Output "Img missing alt: $($imgMissingAlt.Count)"; Write-Output "Empty href: $($emptyHref.Count)"

$reportDir = Join-Path $root 'link-check-report'; if (-not (Test-Path $reportDir)) { New-Item -ItemType Directory -Path $reportDir | Out-Null }
$relativeMissing | ConvertTo-Json -Depth 5 | Out-File (Join-Path $reportDir 'relativeMissing.json')
$anchorMissing | ConvertTo-Json -Depth 5 | Out-File (Join-Path $reportDir 'anchorMissing.json')
$imgMissingAlt | ConvertTo-Json -Depth 5 | Out-File (Join-Path $reportDir 'imgMissingAlt.json')
$externalResults | ConvertTo-Json -Depth 5 | Out-File (Join-Path $reportDir 'externalResults.json')
Write-Output "JSON reports written to $reportDir"
